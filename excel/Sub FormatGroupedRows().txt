Sub FormatGroupedRows()

Dim ws As Worksheet
Dim lastRow As Long
Dim lastCol As Long
Dim r As Long
Dim groupNumber As Long
Dim previousValue As String
Dim currentValue As String
Dim colorIndex As Long

Set ws = ActiveSheet

' Find the last row containing data in Column A
lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row

' Find the last column containing a header in Row 1
lastCol = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column

' Start with the first color pair
groupNumber = 0
previousValue = ""

' Start at row 2 so the header row is ignored
For r = 2 To lastRow

    currentValue = Trim(ws.Cells(r, 1).Value)

    ' Detect when the value in Column A changes
    If currentValue <> previousValue Then
        groupNumber = groupNumber + 1
        previousValue = currentValue
    End If

    ' Cycle through color pairs
    colorIndex = ((groupNumber - 1) Mod 4) + 1

    ' Format only through the last header column
    
If groupNumber Mod 2 = 1 Then

    ' Odd-numbered group = green
    If r Mod 2 = 1 Then
        ws.Range(ws.Cells(r, 1), ws.Cells(r, lastCol)).Interior.Color = RGB(226, 239, 218)
    Else
        ws.Range(ws.Cells(r, 1), ws.Cells(r, lastCol)).Interior.Color = RGB(198, 224, 180)
    End If

Else

    ' Even-numbered group = yellow
    If r Mod 2 = 1 Then
        ws.Range(ws.Cells(r, 1), ws.Cells(r, lastCol)).Interior.Color = RGB(255, 242, 204)
    Else
        ws.Range(ws.Cells(r, 1), ws.Cells(r, lastCol)).Interior.Color = RGB(255, 230, 153)
    End If

End If

Next r

' Format Doniphan County Total rows
For r = 2 To lastRow

    If Trim(ws.Cells(r, 1).Value) = "Doniphan County Total" Then

        If r Mod 2 = 1 Then
            ' Odd row
            ws.Range(ws.Cells(r, 1), ws.Cells(r, lastCol)).Interior.Color = RGB(189, 215, 238)
        Else
            ' Even row
            ws.Range(ws.Cells(r, 1), ws.Cells(r, lastCol)).Interior.Color = RGB(221, 235, 247)
        End If

    End If

Next r


' Format Prairie Hills Schools Total rows
For r = 2 To lastRow

    If Trim(ws.Cells(r, 1).Value) = "Prairie Hills Schools Total" Then

        If r Mod 2 = 1 Then
            ' Odd row
            ws.Range(ws.Cells(r, 1), ws.Cells(r, lastCol)).Interior.Color = RGB(248, 203, 173)
        Else
            ' Even row
            ws.Range(ws.Cells(r, 1), ws.Cells(r, lastCol)).Interior.Color = RGB(252, 228, 214)
        End If

    End If

Next r


End Sub
